FIX 1.1: Create Missing Event Class - App\Events\TicketStatusUpdated
---------------------------------------------------------------------

FILE TO CREATE: app/Events/TicketStatusUpdated.php

<?php

declare(strict_types=1);

namespace App\Events;

use App\Models\HelpdeskTicket;
use Illuminate\Broadcasting\Channel;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class TicketStatusUpdated implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public HelpdeskTicket $ticket,
        public string $oldStatus,
        public string $newStatus,
        public ?int $userId = null
    ) {
    }

    public function broadcastOn(): array
    {
        return [
            new Channel('tickets'),
            new Channel('tickets.' . $this->ticket->id),
        ];
    }

    public function broadcastWith(): array
    {
        return [
            'ticket_id' => $this->ticket->id,
            'ticket_number' => $this->ticket->ticket_number,
            'old_status' => $this->oldStatus,
            'new_status' => $this->newStatus,
            'timestamp' => now()->toISOString(),
            'user_id' => $this->userId,
        ];
    }
}

FIX 2.1: Redis/Queue Tests - Environment Detection
---------------------------------------------------------------------

STRATEGY: Add @group annotation and environment checks

FILE TO MODIFY: tests/Feature/Redis/QueueIntegrationTest.php

ADD AT CLASS LEVEL:
use PHPUnit\Framework\Attributes\Group;

#[Group('requires-redis')]
class QueueIntegrationTest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();
        
        if (!$this->isRedisAvailable()) {
            $this->markTestSkipped('Redis is not available in this environment');
        }
    }
    
    private function isRedisAvailable(): bool
    {
        try {
            $redis = app('redis')->connection();
            $redis->ping();
            return true;
        } catch (\Exception $e) {
            return false;
        }
    }
}

FIX 2.2: Horizon/Reverb Environment Tests
---------------------------------------------------------------------

FILES TO MODIFY:
- tests/Feature/Environment/HorizonWSLRedisTest.php
- tests/Feature/Environment/XamppEnvironmentTest.php
- tests/Feature/ReverbConfigurationTest.php

ADD @group annotation:
#[Group('requires-wsl')]
#[Group('environment-specific')]

UPDATE setUp():
protected function setUp(): void
{
    parent::setUp();
    
    if (env('CI') === true || !$this->isWSLEnvironment()) {
        $this->markTestSkipped('This test requires WSL environment');
    }
}

private function isWSLEnvironment(): bool
{
    // Check if running on WSL
    if (DIRECTORY_SEPARATOR === '\\') {
        return false; // Windows
    }
    
    $release = file_get_contents('/proc/version');
    return stripos($release, 'microsoft') !== false;
}
