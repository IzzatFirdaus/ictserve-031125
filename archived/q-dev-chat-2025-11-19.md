        'job_grade' => ['nullable', 'string', 'max:100'],
        'guest_email' => ['required', 'email', 'max:255'],
        'guest_phone' => ['required', 'string', 'max:20'],
        'category_id' => ['required', 'exists:ticket_categories,id'],
